## 1. Mục tiêu

Có dịch vụ backend khởi động được, kết nối PostgreSQL 16 + pgvector và tạo đủ bảng của mô hình dữ liệu tối thiểu bằng migration, làm nền cho T04, T07–T09, T12–T13.

## 2. Phạm vi

**Làm:** dự án Spring Boot, cấu hình theo môi trường, Flyway V1, Testcontainers cho test tích hợp, endpoint sức khỏe, xử lý lỗi chuẩn.

**Không làm:** đăng nhập (T04); logic tải lên/xử lý (T07, T08).

## 4. Đặc tả kỹ thuật

Lược đồ `[Đề xuất]` — tên cột có thể đổi nếu giữ nguyên quan hệ và ràng buộc:

```sql
CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE users (
  id            uuid PRIMARY KEY,
  username      text NOT NULL UNIQUE,
  password_hash text NOT NULL,
  role          text NOT NULL CHECK (role IN ('ADMIN','MEMBER')),
  failed_logins int  NOT NULL DEFAULT 0,
  locked_until  timestamptz,
  created_at    timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE documents (
  id          uuid PRIMARY KEY,
  name        text NOT NULL,
  size_bytes  bigint NOT NULL,
  page_count  int,
  status      text NOT NULL CHECK (status IN ('PROCESSING','READY','ERROR','DELETED')),
  error_code  text,            -- PASSWORD_PROTECTED | SCANNED_NO_TEXT | CORRUPT | TOO_MANY_PAGES
  warning     jsonb,           -- ví dụ {"code":"UNREADABLE_PAGES","pages":[3,7]}
  storage_key text NOT NULL,
  uploaded_by uuid NOT NULL REFERENCES users(id),
  uploaded_at timestamptz NOT NULL DEFAULT now(),
  ready_at    timestamptz
);

CREATE TABLE chunks (
  id          uuid PRIMARY KEY,
  document_id uuid NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
  page        int  NOT NULL CHECK (page >= 1),
  char_start  int  NOT NULL,   -- offset trong văn bản của trang
  char_end    int  NOT NULL,
  content     text NOT NULL,
  embedding   vector(1536) NOT NULL   -- số chiều theo mô hình chọn ở T06
);
CREATE INDEX chunks_doc_ix ON chunks(document_id, page);
CREATE INDEX chunks_embedding_ix ON chunks USING hnsw (embedding vector_cosine_ops);

CREATE TABLE conversations (
  id uuid PRIMARY KEY, owner_id uuid NOT NULL REFERENCES users(id),
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE messages (
  id uuid PRIMARY KEY,
  conversation_id uuid NOT NULL REFERENCES conversations(id),
  question text NOT NULL,
  answer   text,
  result   text CHECK (result IN ('ANSWERED','NOT_FOUND','PARTIAL','STOPPED','INTERRUPTED','TIMEOUT','ERROR')),
  prompt_tokens int, completion_tokens int, cost_usd numeric(10,6),
  created_at timestamptz NOT NULL DEFAULT now(), finished_at timestamptz
);

CREATE TABLE citations (
  id uuid PRIMARY KEY,
  message_id  uuid NOT NULL REFERENCES messages(id) ON DELETE CASCADE,
  idx         int  NOT NULL,                 -- số [n] trong câu trả lời
  document_id uuid NOT NULL REFERENCES documents(id),
  document_name text NOT NULL,               -- sao chép lúc tạo để hiển thị khi tài liệu đã xóa
  page int NOT NULL, char_start int, char_end int,
  UNIQUE (message_id, idx)
);
```

Ghi chú: bảng nhật ký kiểm toán và sự kiện analytics (EV-xxx) được thêm ở T09 (migration V2). `citations.document_name` được sao chép để US "tài liệu đã bị xóa" (T20) vẫn hiện tên.
